# ---- the recompiled game ---------------------------------------------------
# tools/gsp generates C for every module of the original source (plus the
# image tables the original build tools produced). Regenerate with the
# `regen` target after changing the tools.
# A cross build (Android) uses a directory generated by a host build instead: -DWWF_GEN_DIR=<host build>/gen
# -DWWF_REGEN=OFF. No Python is needed then; the directory must hold c/modules.cmake and the tables.
set(WWF_GEN_DIR ${CMAKE_BINARY_DIR}/gen CACHE PATH "The recompiled game (tools/gsp output)")
option(WWF_REGEN "Generate the recompiled game from orig/ (needs Python 3)" ON)
set(WWF_ORIG ${CMAKE_CURRENT_SOURCE_DIR}/orig)
find_package(Python3 COMPONENTS Interpreter QUIET)
set(WWF_REGEN_CMD
  ${CMAKE_COMMAND} -E env
  ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/genimg.py --src ${WWF_ORIG} --out ${WWF_GEN_DIR}
  && ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/gsp2c.py --src ${WWF_ORIG} --gen ${WWF_GEN_DIR} --out ${WWF_GEN_DIR})
if(WWF_REGEN AND Python3_FOUND AND EXISTS ${WWF_ORIG}/WRESTLE.CMD)
  # The chair's animation scripts are made from the original source (orig/*SEQ3.ASM) and are not kept in the repository.
  if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/mods/chair/asm/CHAIRSEQ.ASM)
    message(STATUS "Making mods/chair/asm/CHAIRSEQ.ASM from the original source...")
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/mods/chair/mkchair.py
                    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} RESULT_VARIABLE r_chair)
    if(NOT r_chair EQUAL 0)
      message(FATAL_ERROR "mods/chair/mkchair.py failed")
    endif()
  endif()
  # Regenerate when the generator changed since the last run (a stale gen/
  # would silently miss, for example, a new PATCHES entry after a git pull).
  file(GLOB WWF_GEN_TOOLS ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/*.py)
  # What mods add to the build (mods/*/gen.txt and the sources they carry, see
  # tools/gsp/extras.py). Linked in but inactive until the mod switches it on.
  set(WWF_EXTRAS_FILE ${CMAKE_BINARY_DIR}/extras.txt)
  # The machine's own changes (src/wolf/core.gen.txt) are always in; the mods' only with WWF_MODS.
  set(WWF_CORE_GEN ${CMAKE_CURRENT_SOURCE_DIR}/src/wolf/core.gen.txt)
  file(GLOB WWF_CORE_ASM ${CMAKE_CURRENT_SOURCE_DIR}/src/wolf/asm/*)
  set(_ex "include ${WWF_CORE_GEN}\n")
  list(APPEND WWF_GEN_TOOLS ${WWF_CORE_GEN} ${WWF_CORE_ASM})
  if(WWF_MODS)
    file(GLOB WWF_MOD_GENS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/mods/*/gen.txt)
    file(GLOB_RECURSE WWF_MOD_ASM CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/mods/*/asm/*
         ${CMAKE_CURRENT_SOURCE_DIR}/mods/*/gen/*)
    foreach(g ${WWF_MOD_GENS})
      string(APPEND _ex "include ${g}\n")
    endforeach()
    list(APPEND WWF_GEN_TOOLS ${WWF_MOD_GENS} ${WWF_MOD_ASM})
  endif()
  file(WRITE ${WWF_EXTRAS_FILE} "${_ex}")
  set(WWF_EXTRAS_ARGS --extras ${WWF_EXTRAS_FILE})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${WWF_GEN_TOOLS})
  set(WWF_GEN_STAMP_TEXT "")
  foreach(f ${WWF_GEN_TOOLS})
    file(SHA1 ${f} h)
    string(APPEND WWF_GEN_STAMP_TEXT "${h}")
  endforeach()
  string(SHA1 WWF_GEN_STAMP "${WWF_GEN_STAMP_TEXT}")
  set(WWF_GEN_STAMP_FILE ${WWF_GEN_DIR}/generator.stamp)
  set(WWF_GEN_OLD "")
  if(EXISTS ${WWF_GEN_STAMP_FILE})
    file(READ ${WWF_GEN_STAMP_FILE} WWF_GEN_OLD)
  endif()
  if(NOT EXISTS ${WWF_GEN_DIR}/c/modules.cmake OR NOT "${WWF_GEN_OLD}" STREQUAL "${WWF_GEN_STAMP}")
    message(STATUS "Recompiling the original source (tools/gsp)...")
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/genimg.py
                            --src ${WWF_ORIG} --out ${WWF_GEN_DIR} ${WWF_EXTRAS_ARGS} RESULT_VARIABLE r1)
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/gsp2c.py
                            --src ${WWF_ORIG} --gen ${WWF_GEN_DIR} --out ${WWF_GEN_DIR} ${WWF_EXTRAS_ARGS} RESULT_VARIABLE r2)
    if(r1 EQUAL 0 AND r2 EQUAL 0)
      file(WRITE ${WWF_GEN_STAMP_FILE} "${WWF_GEN_STAMP}")
    endif()
  endif()
  add_custom_target(regen
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/genimg.py --src ${WWF_ORIG} --out ${WWF_GEN_DIR} ${WWF_EXTRAS_ARGS}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp/gsp2c.py --src ${WWF_ORIG} --gen ${WWF_GEN_DIR} --out ${WWF_GEN_DIR} ${WWF_EXTRAS_ARGS}
    COMMENT "Regenerating recompiled code")
elseif(WWF_REGEN AND NOT EXISTS ${WWF_GEN_DIR}/c/modules.cmake)
  message(FATAL_ERROR "The original game is missing: orig/ is not part of this repository. Run tools/fetch_orig.sh "
                      "(see orig/README.md), then run cmake again. (Python 3 is needed too.)")
endif()
