# Four players: in buddy mode the two computer partners (processes 2 and 3) can be played by a
# third and a fourth person. The partners are drones, whose controls are the words DRN_JOY and
# DRN_BUT that drone_main fills in; while the mod is on, drone_main takes them from the mod
# instead (fp_input in asm/FOURPLAYER.ASM) and then does what it always does with them.
srcdir asm
module FOURPLAYER
edit	DRONE.ASM	after	^\tmove\t\*a13\(DRN_JOY\),a7\t;A7=Old joy$	\tcalla\tfp_input\n\tmove\ta0,a0\n\tjrz\t#fpno\n\tjauc\t#x\n#fpno
edit	SELECT.ASM	after	^\t\.include\t"plyr\.equ"$	\t.def\tcall_wrestler_name
# The select screen: a process draws a square for players 3 and 4, who pick a wrestler for their partner
# (fp_pick3/4, used by choose_buddies); the screen waits a little for a player who has started to choose.

edit	SELECT.ASM	after	^;Players have chosen their guys\.  wait a sec, then bail$	\tmovi\t600,a8\n#fpw\tSLEEPK\t1\n\tmove\t@fp_wait,a14\n\tjrz\t#fpwd\n\tdsj\ta8,#fpw\n#fpwd

# With three people a wrestler slot is empty (process_ptrs entry 0, the partner nobody plays is not made): the rumble
# camera's live count read PLYRMODE, I_WILL_DIE and PLYRNUM at address 0 + offset, which is the top of the video
# bitmap: it counted a wrestler that is not there, with a number taken from picture data, and the camera ran off to
# wherever that pointed (when a wrestler jumped into the top row, the picture there changed). Skip empty slots.
edit	WRESTLE2.ASM	after	^#lp3\tmove\t\*a0\+,a2,L\s*$	\tjrz\t#nxt3
