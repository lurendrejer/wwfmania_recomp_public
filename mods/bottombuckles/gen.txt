# Bottom turnbuckles (mods/bottombuckles/README.md): the game only lets a wrestler climb the two turnbuckles at the far
# edge of the ring (climb_turnbuckle wants UP + LEFT/RIGHT at the top rope). With the mod, DOWN + LEFT/RIGHT at the
# near edge climbs the near ones with the same animations.
srcdir asm
module BOTTOMBUCKLES
# the crowd fence's back corners first (fc_check, asm/BOTTOMBUCKLES.ASM): Up + Left/Right there climbs like a far turnbuckle
edit	WRESTLE2.ASM	after	^ SUBR\tclimb_turnbuckle$	\tmove\t@bb_on,a14\n\tjrz\t#fcno\n\tPUSH\ta0\n\tcalla\tfc_check\n\tmove\ta0,a14\n\tPULL\ta0\n\tmove\ta14,a14\n\tjrz\t#fcno\n\tjauc\t#climbit\n#fcno
# not at the top edge: try the bottom edge (bb_check, asm/BOTTOMBUCKLES.ASM, returns a3 = the direction to face, a0 = 0 if none)
edit	WRESTLE2.ASM	after	^#not_top$	\tmove\t@bb_on,a14\n\tjrz\t#bbno\n\tcallr\tbb_check\n\tmove\ta0,a0\n\tjrz\t#bbno\n\tjauc\t#climbit\n#bbno
# the game moves a near miss onto the corner: the bottom corner for a wrestler at the bottom edge
edit	WRESTLE2.ASM	sub	{1}^\tmovi\tRING_TOP,a14$	\tmovi\tRING_TOP,a14\n\tmove\t*a13(OBJ_ZPOSINT),a0\n\tcmpi\t800,a0\n\tjrge\t#bbn\n\tmovi\tARENA_TOP,a14\n\tjruc\t#bbt\n#bbn\tcmpi\t1600,a0\n\tjrlt\t#bbn2\n\tmovi\tARENA_BOT,a14\n\tjruc\t#bbt\n#bbn2\tcmpi\t(RING_TOP+RING_BOT)/2,a0\n\tjrlt\t#bbt\n\tmovi\tRING_BOT,a14\n#bbt
# the leap up onto the nearest turnbuckle: the corner is in the rope table's header (top: words 0 and 1, bottom: 2 and 3)
edit	SHNSEQ2.ASM	sub	^\tmove\t\*a0\+,a1,W\t\t;X$	\tmove\t@bb_on,a14\n\tjrz\t#bbl\n\tmove\t*a13(OBJ_ZPOSINT),a14\n\tcmpi\t800,a14\n\tjrlt\t#bbf\n\tcmpi\t1600,a14\n\tjrge\t#bbf\n\tcmpi\t(RING_TOP+RING_BOT)/2,a14\n\tjrlt\t#bbl\n\taddk\t32,a0\n\tjruc\t#bbl\n#bbf\tcalla\tfc_table\n#bbl\n\tmove\t*a0+,a1,W\t\t;X
