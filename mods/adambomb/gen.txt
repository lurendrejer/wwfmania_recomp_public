# Every table by wrestler gets Adam's entry from Razor's, renamed to Adam's copy where there is one.
# Slot 7 is never chosen in the original game (the select screen has no square for it), so none of
# this changes the game while the mod is off. Lines marked for Adam keep what the next edits set.
edit	PROGRESS.ASM	sub	^\t\.long\tDNKBLU_P\t;7\t\(spare\)$	\t.long\tADMRED_P\t;7\t(A. Bomb)
edit	DCSSOUND.ASM	sub	^\t\.WORD\s+0\t;7\tspare$	\t.WORD  20DH\t;7\t(A. Bomb: a wail of the others, not Razor's)
edit	DCSSOUND.ASM	sub	^(\t\.word\t0000h,0\t\t;)ADAM BOMB$	\1A. BOMB (Razor's voice is his own)
edit	*	slot	7=1!@(?i)spare|adam|unused	rzr_choking_anim=adm_choking_anim!,rzr_stand_anim=adm_stand_anim!,rzr_fdizzy_anim=adm_fdizzy_anim!,rzr_=adm_,razor_=adam_,move_razor=move_adam,R*=AB*
edit	*	slot	Adam=Razor!	rzr_choking_anim=adm_choking_anim!,rzr_stand_anim=adm_stand_anim!,rzr_fdizzy_anim=adm_fdizzy_anim!,rzr_=adm_,razor_=adam_,move_razor=move_adam,R*=AB*
edit	*	slot	spare=razor!	rzr_choking_anim=adm_choking_anim!,rzr_stand_anim=adm_stand_anim!,rzr_fdizzy_anim=adm_fdizzy_anim!,rzr_=adm_,razor_=adam_,move_razor=move_adam,R*=AB*
# animations of Razor as the victim that live in another wrestler's file get a copy for Adam there
# (the names in the renames above): the Undertaker's choke, and the dead wrestler standing up dizzy
edit	UNDSEQ3.ASM	cloneblock	^rzr_choking_anim\s*$:^und_choking_anim\s*$	rzr_=adm_!,R*=AB*
edit	FINISEQ.ASM	cloneblock	^ SUBRP\trzr_stand_anim\s*$:^ SUBRP\thrt_stand_anim\s*$	rzr_=adm_!,R*=AB*
# entries without a comment of their own: between Doink's and Lex's, from Razor's six before
edit	*	tokslot	6:^#?(dnk|doink):^#?lex:^#?(rzr_|razor_|move_razor$|rdizzy)	rzr_choking_anim=adm_choking_anim!,rzr_stand_anim=adm_stand_anim!,rzr_fdizzy_anim=adm_fdizzy_anim!,rzr_=adm_,razor_=adam_,move_razor=move_adam,R*=AB*
edit	*	copyblock	^#Adam\s*$=^#Razor\s*$	rzr_choking_anim=adm_choking_anim!,rzr_stand_anim=adm_stand_anim!,rzr_fdizzy_anim=adm_fdizzy_anim!,rzr_=adm_,razor_=adam_,move_razor=move_adam,R*=AB*
edit	TABLES.ASM	copyblock	^;7 Adam Bomb=^;1 Razor Ramon	rzr_choking_anim=adm_choking_anim!,rzr_stand_anim=adm_stand_anim!,rzr_fdizzy_anim=adm_fdizzy_anim!,rzr_=adm_,razor_=adam_,move_razor=move_adam,R*=AB*
# his eight palettes (by the button held when he is chosen) and those of buddy mode (blue, red)
edit	WRESTLE2.ASM	sub	{1}^\t\.long\t0,0,0,0,0,0,0,0\s*$	\t.long\tADMRED_P,ADMGRN_P,ADMYEL_P,ADMORG_P,ADMPNK_P,ADMBLU_P,ADMPRP_P,ADMGRY_P\t;(A. Bomb)
edit	WRESTLE2.ASM	sub	{1}^\t\.long\t0,0\s*$	\t.long\tADMBLU_P,ADMRED_P

# first that is still marked spare
edit	PROGRESS.ASM	sub	{1}^\t\.long\t0,0\t\t;7 spare$	\t.long\tadm_pg_taunt_anim,adm_torso4_anim\t;7 (A. Bomb)
